G4ThreeVector ActGeant::PrimaryGenerator::RotateToWorld ( const G4ThreeVector &  vec,
const G4ThreeVector &  beamDir 
)
private
286{
287
288 // Usually a sampled reaction returns the angles in the beam frame
289 // But to compute geometrical things (propagate that track to a silicon)
290 // We need to work in the "geometry = world" frame,
291 // where in ACTAR the "beam" goes in {1, 0, 0}
292 // Using XYZPoint and XYZVector is easy to compute it,
293 // as quoted here: https://root-forum.cern.ch/t/get-3x3-rotation-matrix-between-two-tvector3/60070
294 // Using only GenVector classes
295 auto originalFrame {beamDir.unit()};
296 G4ThreeVector worldFrame {1, 0, 0};
297 auto cross {worldFrame.cross(originalFrame)}; // this defines the rotation axis
298 auto angle {std::acos(originalFrame.dot(worldFrame))};
299 ROOT::Math::AxisAngle axis {cross, angle};
300 ROOT::Math::Rotation3D rotation {axis};
301 return rotation(vec);
302}
