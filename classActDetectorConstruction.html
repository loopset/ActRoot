263{
264 // Retrieve file path. So far name is hardcoded
265 auto file {ActRoot::Options::GetInstance()->GetConfigDir() + "/geant.conf"};
266 ActRoot::InputParser parser {file};
267
268 // Get [Gas] block
269 auto gas {parser.GetBlock("Gas")};
270 // Gas components
271 auto comps {gas->GetStringVector("Components")};
272 // VOLUME = MOLE fractions
273 auto fracs {gas->GetDoubleVector("Fractions")};
274 // Pressure
275 auto p {gas->GetDouble("Pressure")}; // in mbar
276 p *= 1e-3 * bar;
277
278 // Gas constant
279 double R {8.314 * joule / (mole * kelvin)};
280 // Temperature (20C assumed)
281 double t {293.15 * kelvin};
282
283 // Pointer to NIST
284 auto* nist {G4NistManager::Instance()};
285
286 // Build materials of mixture
287 std::vector<G4Material*> mats {};
288 std::vector<double> molarMasses {};
289 for(const auto& comp : comps)
290 {
291 G4Material* mat {};
292 double molarMass {};
293 double density {};
294 if(comp == "D2")
295 {
296 molarMass = 2. * 2.0140 * g / mole;
297 density = (p * molarMass) / (R * t);
298 mat = new G4Material(comp, density, 1, kStateGas, t, p);
299 mat->AddElement(G4Element::GetElement("D"), 2);
300 }
301 if(comp == "iC4H10")
302 {
303 molarMass = (4. * 12.0107 + 10. * 1.00794) * g / mole;
304 density = (p * molarMass) / (R * t);
305 mat = new G4Material {"iC4H10", density, 2, kStateGas, t, p};
306 mat->AddElement(nist->FindOrBuildElement("C"), 4);
307 mat->AddElement(nist->FindOrBuildElement("H"), 10);
308 }
309 std::cout << "Comp : " << comp << " molar mass: " << molarMass / (g / mole) << " rho: " << density / (g / cm3)
310 << '\n';
311 mats.push_back(mat);
312 molarMasses.push_back(molarMass);
313 }
314 // Compute total mass of mixture
315 double massMix {};
316 for(int i = 0; i < molarMasses.size(); i++)
317 massMix += static_cast<double>(fracs[i]) / 100 * molarMasses[i];
318 // And mass fractions
319 std::vector<double> massFracs {};
320 for(int i = 0; i < molarMasses.size(); i++)
321 massFracs.push_back((static_cast<double>(fracs[i]) / 100 * molarMasses[i]) / massMix);
322
323 // And density
324 auto rhoMix {(p * massMix) / (R * t)};
325
326 std::cout << "Rho of mixture : " << rhoMix / (g / cm3) << " g/cm3" << '\n';
327 auto* mixture {new G4Material {"GasMixture", rhoMix, static_cast<G4int>(mats.size()), kStateGas, t, p}};
328 for(int i = 0; i < mats.size(); i++)
329 {
330 const auto& mat {mats[i]};
331 mixture->AddMaterial(mat, massFracs[i]);
332 }
333}
