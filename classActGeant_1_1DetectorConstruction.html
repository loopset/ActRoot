274{
275 // Retrieve file path. So far name is hardcoded
276 auto file {ActRoot::Options::GetInstance()->GetConfigDir() + "/geant.conf"};
277 ActRoot::InputParser parser {file};
278
279 // Get [Gas] block
280 auto gas {parser.GetBlock("Gas")};
281 // Gas components
282 auto comps {gas->GetStringVector("Components")};
283 // VOLUME = MOLE fractions
284 auto fracs {gas->GetDoubleVector("Fractions")};
285 // Pressure
286 auto p {gas->GetDouble("Pressure")}; // in mbar
287 p *= 1e-3 * bar;
288
289 // Gas constant
290 double R {8.314 * joule / (mole * kelvin)};
291 // Temperature (20C assumed)
292 double t {293.15 * kelvin};
293
294 // Pointer to NIST
295 auto* nist {G4NistManager::Instance()};
296
297 // Build materials of mixture
298 std::vector<G4Material*> mats {};
299 std::vector<double> molarMasses {};
300 for(const auto& comp : comps)
301 {
302 G4Material* mat {};
303 double molarMass {};
304 double density {};
305 if(comp == "D2")
306 {
307 molarMass = 2. * 2.0140 * g / mole;
308 density = (p * molarMass) / (R * t);
309 mat = new G4Material(comp, density, 1, kStateGas, t, p);
310 mat->AddElement(G4Element::GetElement("D"), 2);
311 }
312 if(comp == "iC4H10")
313 {
314 molarMass = (4. * 12.0107 + 10. * 1.00794) * g / mole;
315 density = (p * molarMass) / (R * t);
316 mat = new G4Material {"iC4H10", density, 2, kStateGas, t, p};
317 mat->AddElement(nist->FindOrBuildElement("C"), 4);
318 mat->AddElement(nist->FindOrBuildElement("H"), 10);
319 }
320 std::cout << "Comp : " << comp << " molar mass: " << molarMass / (g / mole) << " rho: " << density / (g / cm3)
321 << '\n';
322 mats.push_back(mat);
323 molarMasses.push_back(molarMass);
324 }
325 // Compute total mass of mixture
326 double massMix {};
327 for(int i = 0; i < molarMasses.size(); i++)
328 massMix += static_cast<double>(fracs[i]) / 100 * molarMasses[i];
329 // And mass fractions
330 std::vector<double> massFracs {};
331 for(int i = 0; i < molarMasses.size(); i++)
332 massFracs.push_back((static_cast<double>(fracs[i]) / 100 * molarMasses[i]) / massMix);
333
334 // And density
335 auto rhoMix {(p * massMix) / (R * t)};
336
337 std::cout << "Rho of mixture : " << rhoMix / (g / cm3) << " g/cm3" << '\n';
338 auto* mixture {new G4Material {"GasMixture", rhoMix, static_cast<G4int>(mats.size()), kStateGas, t, p}};
339 for(int i = 0; i < mats.size(); i++)
340 {
341 const auto& mat {mats[i]};
342 mixture->AddMaterial(mat, massFracs[i]);
343 }
